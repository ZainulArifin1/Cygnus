Should I propose that my love be but a fleeting fancy, let it be known, my dear, that I reject that null hypothesis with fervor, for my affection for thee is a truth beyond all reasonable doubt! (Hypothesis testing)

# Set seed
set.seed(123)

# Generate two normal distributions
n <- 100
dist1 <- rnorm(n, mean = 5, sd = 1)
dist2 <- rnorm(n, mean = 7, sd = 1)

data <- data.frame(
  value = c(dist1, dist2),
  group = rep(c("I dislike you", "Ground Truth"), each = n)
)

# Perform a t-test to compare the means
t_test_result <- t.test(dist1, dist2)
p_value <- t_test_result$p.value

# Create the plot
ggplot(data, aes(x = value, fill = group)) +
  geom_density(alpha = 0.5) +
  geom_vline(aes(xintercept = mean(dist1)), color = "blue", linetype = "dashed", size = 1) +
  geom_vline(aes(xintercept = mean(dist2)), color = "red", linetype = "dashed", size = 1) +
  labs(title = paste("Comparison of Two Distributions\n(p-value =", p_value, ")"),
       x = "Value",
       y = "Density") +
  theme_minimal() +
  scale_fill_manual(values = c("I dislike you" = "blue", "Ground Truth" = "red")) +
  annotate("text", x = 5, y = 0.3, label = paste("Mean 1:", round(mean(dist1), 2)), color = "black", size = 3) +
  annotate("text", x = 7, y = 0.3, label = paste("Mean 2:", round(mean(dist2), 2)), color = "black", size = 3)

set.seed(123)
n <- 50  

cygnus_data <- data.frame(
  Group = rep("Cygnus", n),
  PC1 = rnorm(n, mean = 5, sd = 1),
  PC2 = rnorm(n, mean = 5, sd = 1)
)
other_data <- data.frame(
  Group = rep("Other", n),
  PC1 = rnorm(n, mean = 0, sd = 1),
  PC2 = rnorm(n, mean = 0, sd = 1)
)
stuffs_data <- data.frame(
  Group = rep("Stuffs", n),
  PC1 = rnorm(n, mean = -5, sd = 1),
  PC2 = rnorm(n, mean = -5, sd = 1)
)
all_data <- rbind(cygnus_data, other_data, stuffs_data)

# Scale the data (excluding the Group column)
scaled_data <- scale(all_data[, 2:3])

# Perform PCA
pca_result <- prcomp(scaled_data, center = TRUE, scale. = TRUE)
pca_df <- data.frame(pca_result$x)
pca_df$Group <- all_data$Group

# Plot the PCA results
ggplot(pca_df, aes(x = PC1, y = PC2, color = Group)) +
  geom_point(size = 3) +
  labs(title = "PCA of Generated Data",
       x = "Principal Component 1",
       y = "Principal Component 2") +
  theme_minimal()
